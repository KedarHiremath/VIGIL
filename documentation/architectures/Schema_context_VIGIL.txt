"""Derived media artifact schema."""

import re
from datetime import UTC, datetime
from typing import Literal

from pydantic import Field, field_validator, model_validator

from vigil.domain.base import DomainModel
from vigil.domain.common import Identifier, Timestamp
from vigil.domain.source_reference import SourceReference


class Artifact(DomainModel):
    """Represents media derived from an original source."""

    artifact_id: Identifier = Field(
        description="Stable unique identifier for the derived artifact.",
    )

    artifact_type: Literal[
        "frame",
        "crop",
        "clip",
        "audio_segment",
        "derived_media",
    ] = Field(
        description="Operation-specific kind of derived media.",
    )

    media_type: Literal["video", "audio", "image"] = Field(
        description="Actual media type stored by this artifact.",
    )

    storage_uri: str = Field(
        min_length=1,
        description="Storage location of the derived media.",
    )

    source_reference: SourceReference = Field(
        description=(
            "Original media and timeline interval from which this artifact was derived."
        ),
    )

    sha256: str | None = Field(
        default=None,
        description=(
            "Optional SHA-256 checksum of the derived artifact, "
            "represented by exactly 64 hexadecimal characters."
        ),
    )

    created_at: Timestamp = Field(
        default_factory=lambda: datetime.now(UTC),
        description="Timezone-aware timestamp when the artifact was created.",
    )

    @field_validator("storage_uri")
    @classmethod
    def validate_storage_uri(cls, value: str) -> str:
        """Reject blank artifact locations."""
        if not value.strip():
            raise ValueError("Storage URI must not be blank.")
        return value

    @field_validator("sha256")
    @classmethod
    def validate_sha256(cls, value: str | None) -> str | None:
        """Validate the optional SHA-256 checksum."""
        if value is not None and not re.fullmatch(r"[0-9a-fA-F]{64}", value):
            raise ValueError("SHA-256 must contain 64 hexadecimal characters.")
        return value

    @model_validator(mode="after")
    def validate_artifact_type(self) -> "Artifact":
        """Ensure the artifact kind matches its media type."""
        expected_types = {
            "frame": "image",
            "crop": "image",
            "clip": "video",
            "audio_segment": "audio",
        }

        expected = expected_types.get(self.artifact_type)
        if expected is not None and self.media_type != expected:
            raise ValueError(
                f"Artifact type '{self.artifact_type}' "
                f"requires media type '{expected}'."
            )

        return self

import math
from zoneinfo import ZoneInfo, ZoneInfoNotFoundError

from pydantic import Field, field_validator

from vigil.domain.base import DomainModel
from vigil.domain.common import Identifier


class Camera(DomainModel):
    """Represents a camera or recording source."""

    camera_id: Identifier = Field(
        description="Stable unique identifier for this camera.",
    )

    name: str = Field(
        min_length=1,
        description="Human-readable name assigned to the camera.",
    )

    location_label: str | None = Field(
        default=None,
        description="Optional physical location, such as Parking Entrance.",
    )

    timezone: str | None = Field(
        default=None,
        description="Optional IANA timezone, such as Asia/Kolkata.",
    )

    clock_offset_seconds: float | None = Field(
        default=None,
        description=(
            "Optional correction in seconds applied to the camera's "
            "reported clock to estimate reference time."
        ),
    )

    @field_validator("timezone")
    @classmethod
    def validate_timezone(cls, value: str | None) -> str | None:
        """Reject unknown IANA timezone names."""
        if value is None:
            return None

        try:
            ZoneInfo(value)
        except (ZoneInfoNotFoundError, ValueError) as exc:
            raise ValueError("Unknown IANA timezone.") from exc

        return value

    @field_validator("clock_offset_seconds")
    @classmethod
    def validate_clock_offset(cls, value: float | None) -> float | None:
        """Reject infinite and NaN clock offsets."""
        if value is not None and not math.isfinite(value):
            raise ValueError("Clock offset must be finite.")

        return value

src\vigil\domain\common.py
"""Reusable types and validation rules for VIGIL domain schemas."""

from typing import Annotated

from pydantic import (
    AfterValidator,
    AwareDatetime,
    Field,
    StringConstraints,
)

Identifier = Annotated[
    str,
    StringConstraints(
        strip_whitespace=True,
        min_length=1,
    ),
]
"""A nonblank string identifier for a VIGIL object."""


Timestamp = AwareDatetime
"""A datetime that must include timezone information."""


ConfidenceScore = Annotated[
    float,
    Field(
        ge=0.0,
        le=1.0,
        allow_inf_nan=False,
    ),
]
"""An optional-to-use, normalized confidence score between 0 and 1."""


def validate_unique_reference_ids(values: list[str]) -> list[str]:
    """Reject blank or duplicate identifiers in a reference list."""
    normalized = [value.strip() for value in values]

    if any(not value for value in normalized):
        raise ValueError("Reference IDs must not be blank")

    if len(normalized) != len(set(normalized)):
        raise ValueError("Reference IDs must be unique")

    return normalized


ReferenceIds = Annotated[
    list[Identifier],
    AfterValidator(validate_unique_reference_ids),
]
"""A list of nonblank, unique object reference IDs."""

"""Domain schema for investigation coverage."""

from enum import Enum

from pydantic import Field, field_validator, model_validator

from vigil.domain.base import DomainModel
from vigil.domain.common import Identifier, ReferenceIds, Timestamp


class CoverageStatus(str, Enum):
    """Outcome of examining a source or time interval."""

    EXAMINED = "examined"
    PARTIAL = "partial"
    FAILED = "failed"
    NOT_EXAMINED = "not_examined"


class CoverageRecord(DomainModel):
    """Records the examination status of a source or interval."""

    coverage_id: Identifier = Field(min_length=1)
    investigation_id: Identifier = Field(min_length=1)
    source_reference_id: Identifier = Field(min_length=1)

    status: CoverageStatus

    interval_start: Timestamp | None = None
    interval_end: Timestamp | None = None
    examined_at: Timestamp | None = None

    notes: str | None = None
    artifact_ids: ReferenceIds = Field(default_factory=list)

    @field_validator("notes")
    @classmethod
    def reject_blank_notes(cls, value: str | None) -> str | None:
        """Reject blank notes when supplied."""
        if value is not None and not value.strip():
            raise ValueError("notes must not be blank")
        return value

    @model_validator(mode="after")
    def validate_interval_and_status(self) -> "CoverageRecord":
        """Validate interval order and examination timestamps."""
        if (self.interval_start is None) != (self.interval_end is None):
            raise ValueError(
                "interval_start and interval_end must be provided together"
            )

        if (
            self.interval_start is not None
            and self.interval_end is not None
            and self.interval_end < self.interval_start
        ):
            raise ValueError("interval_end must not precede interval_start")

        if self.status == CoverageStatus.EXAMINED and self.examined_at is None:
            raise ValueError("examined status requires examined_at")

        return self

import math

from pydantic import Field, field_validator, model_validator

from vigil.domain.base import DomainModel
from vigil.domain.common import Identifier, Timestamp


class SourceReference(DomainModel):
    """Identifies a precise interval in an original media asset."""

    media_asset_id: Identifier = Field(
        description="ID of the original media containing the evidence.",
    )

    start_seconds: float = Field(
        ge=0,
        description=(
            "Inclusive start position in seconds on the original media timeline."
        ),
    )

    end_seconds: float = Field(
        gt=0,
        description=(
            "Exclusive end position in seconds on the original media timeline."
        ),
    )

    source_timestamp: Timestamp | None = Field(
        default=None,
        description=(
            "Optional timezone-aware wall-clock timestamp corresponding "
            "to start_seconds; retain its timezone or normalize to UTC."
        ),
    )

    @field_validator("start_seconds", "end_seconds")
    @classmethod
    def validate_finite_time(cls, value: float) -> float:
        """Reject NaN and infinite timeline positions."""
        if not math.isfinite(value):
            raise ValueError("Timeline positions must be finite.")
        return value

    @model_validator(mode="after")
    def validate_interval(self) -> "SourceReference":
        """Require a positive-length half-open interval."""
        if self.end_seconds <= self.start_seconds:
            raise ValueError("End time must be greater than start time.")
        return self

"""Runtime state of a VIGIL investigation workflow."""

from datetime import UTC, datetime
from enum import Enum

from pydantic import Field, model_validator

from vigil.domain.base import DomainModel
from vigil.domain.common import Identifier, ReferenceIds, Timestamp


class InvestigationStatus(str, Enum):
    """Lifecycle status of an investigation workflow."""

    CREATED = "created"
    RUNNING = "running"
    PAUSED = "paused"
    COMPLETED = "completed"
    FAILED = "failed"


class InvestigationState(DomainModel):
    """Tracks progress and outputs of an investigation workflow."""

    investigation_id: Identifier = Field(
        description="Stable identifier for the investigation.",
    )

    status: InvestigationStatus = Field(
        default=InvestigationStatus.CREATED,
        description="Current lifecycle status of the investigation.",
    )

    task_ids: ReferenceIds = Field(
        default_factory=list,
        description="IDs of tasks associated with this investigation.",
    )

    observation_ids: ReferenceIds = Field(
        default_factory=list,
        description="IDs of observations produced during the investigation.",
    )

    event_ids: ReferenceIds = Field(
        default_factory=list,
        description="IDs of events identified during the investigation.",
    )

    started_at: Timestamp | None = Field(
        default=None,
        description="Optional timezone-aware time when the investigation began.",
    )

    updated_at: Timestamp = Field(
        default_factory=lambda: datetime.now(UTC),
        description="Timezone-aware time when the state was last updated.",
    )

    completed_at: Timestamp | None = Field(
        default=None,
        description=(
            "Timezone-aware time when the investigation reached a terminal state."
        ),
    )

    @model_validator(mode="after")
    def validate_state(self) -> "InvestigationState":
        """Ensure investigation status and timestamps are consistent."""
        if (
            self.status
            in {
                InvestigationStatus.COMPLETED,
                InvestigationStatus.FAILED,
            }
            and self.completed_at is None
        ):
            raise ValueError("Completed or failed investigations require completed_at.")

        if (
            self.started_at is not None
            and self.completed_at is not None
            and self.completed_at < self.started_at
        ):
            raise ValueError("Completion time cannot precede investigation start time.")

        return self

"""Shared task and model-output contracts for VIGIL."""

import math
from datetime import UTC, datetime
from enum import Enum

from pydantic import Field, JsonValue, field_validator, model_validator

from vigil.domain.base import DomainModel
from vigil.domain.common import (
    ConfidenceScore,
    Identifier,
    ReferenceIds,
    Timestamp,
)


class TaskStatus(str, Enum):
    """Lifecycle status of an execution task."""

    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    PARTIAL = "partial"
    FAILED = "failed"


class ErrorDetail(DomainModel):
    """Structured information describing an execution error."""

    code: str = Field(
        min_length=1,
        description="Stable machine-readable error code.",
    )

    message: str = Field(
        min_length=1,
        description="Human-readable explanation of the error.",
    )

    retryable: bool = Field(
        description="Whether retrying the operation may succeed.",
    )

    retry_after_seconds: float | None = Field(
        default=None,
        ge=0,
        description="Optional minimum delay before retrying, in seconds.",
    )

    @field_validator("retry_after_seconds")
    @classmethod
    def validate_retry_delay(cls, value: float | None) -> float | None:
        """Reject non-finite retry delays."""
        if value is not None and not math.isfinite(value):
            raise ValueError("Retry delay must be finite.")
        return value

    @field_validator("code", "message")
    @classmethod
    def validate_nonblank_text(cls, value: str) -> str:
        """Reject blank error codes and messages."""
        if not value.strip():
            raise ValueError("Error text must not be blank.")
        return value

    @model_validator(mode="after")
    def validate_retry_policy(self) -> "ErrorDetail":
        """Allow retry delays only for retryable errors."""
        if self.retry_after_seconds is not None and not self.retryable:
            raise ValueError("A retry delay requires a retryable error.")
        return self


class TaskRequest(DomainModel):
    """Defines a request for a tool or agent to perform work."""

    task_id: Identifier = Field(
        description="Stable unique identifier for this task.",
    )

    task_type: str = Field(
        min_length=1,
        description=(
            "Operation to perform, such as frame_sampling or visual_analysis."
        ),
    )

    input_refs: ReferenceIds = Field(
        default_factory=list,
        description=(
            "IDs of the media assets, artifacts or other inputs required for the task."
        ),
    )

    parameters: dict[str, JsonValue] = Field(
        default_factory=dict,
        description="JSON-compatible parameters controlling the operation.",
    )

    requested_by: Identifier | None = Field(
        default=None,
        description="Optional identifier of the agent or component requesting work.",
    )

    created_at: Timestamp = Field(
        default_factory=lambda: datetime.now(UTC),
        description="Timezone-aware time when the task request was created.",
    )

    @field_validator("task_type")
    @classmethod
    def validate_task_type(cls, value: str) -> str:
        """Reject blank task types."""
        if not value.strip():
            raise ValueError("Task type must not be blank.")
        return value


class TaskResult(DomainModel):
    """Records the execution outcome of a task."""

    task_id: Identifier = Field(
        description="ID of the task whose execution produced this result.",
    )

    status: TaskStatus = Field(
        description="Current execution outcome or lifecycle status.",
    )

    output_refs: ReferenceIds = Field(
        default_factory=list,
        description=(
            "IDs of outputs produced by the task, such as observations or artifacts."
        ),
    )

    error: ErrorDetail | None = Field(
        default=None,
        description="Structured failure or partial-execution details, if any.",
    )

    started_at: Timestamp | None = Field(
        default=None,
        description="Optional timezone-aware time when execution began.",
    )

    completed_at: Timestamp | None = Field(
        default=None,
        description="Optional timezone-aware time when execution ended.",
    )

    @model_validator(mode="after")
    def validate_result(self) -> "TaskResult":
        """Keep task status, errors and timestamps consistent."""
        if self.status == TaskStatus.FAILED and self.error is None:
            raise ValueError("A failed task must include error details.")

        if self.status == TaskStatus.COMPLETED and self.error is not None:
            raise ValueError("A completed task must not include an error.")

        if (
            self.started_at is not None
            and self.completed_at is not None
            and self.completed_at < self.started_at
        ):
            raise ValueError("Completion time cannot precede start time.")

        return self


class ModelOutput(DomainModel):
    """Represents structured output from a model or analysis tool."""

    output_id: Identifier = Field(
        description="Stable unique identifier for this output.",
    )

    producer_name: str = Field(
        min_length=1,
        description="Name of the model or analysis tool producing the output.",
    )

    output_type: str = Field(
        min_length=1,
        description=(
            "Kind of output, such as frame_description, detection or transcription."
        ),
    )

    payload: dict[str, JsonValue] = Field(
        description="JSON-compatible structured output produced by the tool.",
    )

    confidence: ConfidenceScore | None = Field(
        default=None,
        description=("Optional producer-reported confidence score between 0 and 1."),
    )

    artifact_ids: ReferenceIds = Field(
        default_factory=list,
        description="IDs of artifacts associated with the output.",
    )

    created_at: Timestamp = Field(
        default_factory=lambda: datetime.now(UTC),
        description="Timezone-aware time when the output was created.",
    )
